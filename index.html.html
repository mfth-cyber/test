// Kalkulasi matematika harga & waktu pengerjaan yang realistis (Standar Overgear) function calculateBoostMetrics(startLvl, endLvl, isExpress, isStream) { let basePrice = 0; let baseHours = 0; // Tentukan starting base price berdasarkan level awal if (startLvl < 10) basePrice += CONFIG.startingBasePrice.tier1; else if (startLvl < 20) basePrice += CONFIG.startingBasePrice.tier10; else if (startLvl < 30) basePrice += CONFIG.startingBasePrice.tier20; else if (startLvl < 40) basePrice += CONFIG.startingBasePrice.tier30; else if (startLvl < 50) basePrice += CONFIG.startingBasePrice.tier40; else basePrice += CONFIG.startingBasePrice.tier50; // Tambahkan kenaikan per level & akumulasi waktu realistis (jam per level) for (let lvl = startLvl; lvl < endLvl; lvl++) { if (lvl < 10) { basePrice += CONFIG.stepIncrements.tier1_10; baseHours += 1.5; // Level awal (1-9) relatif cepat } else if (lvl < 20) { basePrice += CONFIG.stepIncrements.tier10_20; baseHours += 3.0; // Level 10-19 } else if (lvl < 30) { basePrice += CONFIG.stepIncrements.tier20_30; baseHours += 5.0; // Level 20-29 } else if (lvl < 35) { basePrice += CONFIG.stepIncrements.tier30_35; baseHours += 7.5; // Level 30-34 } else if (lvl < 40) { basePrice += CONFIG.stepIncrements.tier35_40; baseHours += 9.0; // Level 35-39 } else if (lvl < 50) { basePrice += CONFIG.stepIncrements.tier40_50; baseHours += 11.0; // Level 40-49 (Mulai berat) } else { basePrice += CONFIG.stepIncrements.tier50_60; baseHours += 16.5; // Level 50-60 (Endgame brutal, total kisaran 14 hari untuk 1-60) } } // Terapkan diskon penurunan base price 10% let discountedBase = basePrice * CONFIG.basePriceDiscount; let multiplier = 1.0; if (isExpress) { multiplier += CONFIG.addons.expressMultiplier; baseHours *= 0.75; // Express memangkas waktu sekitar 25% } if (isStream) { multiplier += CONFIG.addons.streamMultiplier; } const subtotal = discountedBase * multiplier; const finalPrice = subtotal * (1 - (CONFIG.launchDiscountPercent / 100)); return { originalPrice: (subtotal).toFixed(2), price: (finalPrice).toFixed(2), hours: Math.max(2, Math.round(baseHours)) }; } // Format waktu menjadi "[X]d [Y]h" (contoh: 9d 16h) function formatDuration(totalHours) { const days = Math.floor(totalHours / 24); const hours = totalHours % 24; if (days === 0) return `~${hours} Hours`; return hours > 0 ? `~${days}d ${hours}h` : `~${days} Days`; }
